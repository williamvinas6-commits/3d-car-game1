<script src="https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js"></script>

<script>
const scene = new THREE.Scene();
scene.background = new THREE.Color(0x87ceeb);
scene.fog = new THREE.Fog(0x87ceeb,30,150);

const camera = new THREE.PerspectiveCamera(
 70,innerWidth/innerHeight,0.1,200
);
camera.position.set(0,5,10);
camera.lookAt(0,0,0);

const renderer = new THREE.WebGLRenderer({antialias:true});
renderer.setSize(innerWidth,innerHeight);
renderer.setPixelRatio(Math.min(devicePixelRatio,2));
document.getElementById("game").appendChild(renderer.domElement);

scene.add(new THREE.HemisphereLight(0xffffff,0x555555,2));

const light = new THREE.DirectionalLight(0xffffff,2);
light.position.set(5,10,5);
scene.add(light);

// ROAD
const road = new THREE.Mesh(
 new THREE.PlaneGeometry(12,200),
 new THREE.MeshStandardMaterial({color:0x303030})
);
road.rotation.x=-Math.PI/2;
road.position.z=-70;
scene.add(road);

// GRASS
const grass = new THREE.Mesh(
 new THREE.PlaneGeometry(200,200),
 new THREE.MeshStandardMaterial({color:0x228b22})
);
grass.rotation.x=-Math.PI/2;
grass.position.set(0,-0.06,-70);
scene.add(grass);

// ROAD LINES
const lines=[];
for(let i=0;i<30;i++){
 const line=new THREE.Mesh(
  new THREE.BoxGeometry(0.15,0.03,3),
  new THREE.MeshBasicMaterial({color:0xffffff})
 );
 line.position.set(0,0.02,-i*7);
 scene.add(line);
 lines.push(line);
}

// PLAYER CAR
function makeCar(color){
 const car=new THREE.Group();

 const body=new THREE.Mesh(
  new THREE.BoxGeometry(1.6,0.65,2.8),
  new THREE.MeshStandardMaterial({color,metalness:0.5,roughness:0.3})
 );
 body.position.y=0.65;
 car.add(body);

 const top=new THREE.Mesh(
  new THREE.BoxGeometry(1.15,0.65,1.35),
  new THREE.MeshStandardMaterial({color:0x222222})
 );
 top.position.set(0,1.2,-0.15);
 car.add(top);

 for(let x of [-0.85,0.85]){
  for(let z of [-0.85,0.85]){
   const wheel=new THREE.Mesh(
    new THREE.CylinderGeometry(0.35,0.35,0.2,16),
    new THREE.MeshStandardMaterial({color:0x111111})
   );
   wheel.rotation.z=Math.PI/2;
   wheel.position.set(x,0.4,z);
   car.add(wheel);
  }
 }
 return car;
}

const player=makeCar(0xff0000);
player.position.set(0,0,5);
scene.add(player);

let enemies=[];
function createEnemy(){
 const enemy=makeCar(
  [0x0000ff,0xffff00,0xffffff,0xff8800][Math.floor(Math.random()*4)]
 );
 enemy.position.set([-3,-1.5,1.5,3][Math.floor(Math.random()*4)],0,-90);
 scene.add(enemy);
 enemies.push(enemy);
}

// TREES
let trees=[];
function createTree(x,z){
 const tree=new THREE.Group();

 const trunk=new THREE.Mesh(
  new THREE.CylinderGeometry(0.2,0.3,2),
  new THREE.MeshStandardMaterial({color:0x754321})
 );
 trunk.position.y=1;
 tree.add(trunk);

 const leaves=new THREE.Mesh(
  new THREE.ConeGeometry(1.3,3,8),
  new THREE.MeshStandardMaterial({color:0x006400})
 );
 leaves.position.y=3;
 tree.add(leaves);

 tree.position.set(x,0,z);
 scene.add(tree);
 trees.push(tree);
}

for(let i=0;i<35;i++){
 createTree(-10, -i*6);
 createTree(10, -i*6);
}

let running=false;
let score=0;
let speed=0;
let lane=0;
let spawnTimer=0;

function moveLeft(){lane=Math.max(-3,lane-1.5);}
function moveRight(){lane=Math.min(3,lane+1.5);}

document.getElementById("left").addEventListener("pointerdown",moveLeft);
document.getElementById("right").addEventListener("pointerdown",moveRight);

window.addEventListener("keydown",e=>{
 if(e.key==="ArrowLeft" || e.key==="a")moveLeft();
 if(e.key==="ArrowRight" || e.key==="d")moveRight();
});

function startGame(){
 enemies.forEach(e=>scene.remove(e));
 enemies=[];
 lane=0;
 score=0;
 speed=0;
 player.position.x=0;
 running=true;
 document.getElementById("start").style.display="none";
}

function endGame(){
 running=false;
 document.getElementById("start").innerText="GAME OVER - RESTART";
 document.getElementById("start").style.display="block";
}

function animate(){
 requestAnimationFrame(animate);

 if(running){
  speed=Math.min(speed+0.15,220);
  score+=1;

  player.position.x+=(lane-player.position.x)*0.15;

  lines.forEach(line=>{
   line.position.z+=0.8;
   if(line.position.z>10)line.position.z=-200;
  });

  trees.forEach(tree=>{
   tree.position.z+=0.8;
   if(tree.position.z>12)tree.position.z=-200;
  });

  spawnTimer++;
  if(spawnTimer>75){
   createEnemy();
   spawnTimer=0;
  }

  enemies.forEach((enemy,i)=>{
   enemy.position.z+=0.8;

   if(Math.abs(enemy.position.x-player.position.x)<1.4 &&
      Math.abs(enemy.position.z-player.position.z)<2.2){
    endGame();
   }

   if(enemy.position.z>15){
    scene.remove(enemy);
    enemies.splice(i,1);
   }
  });

  document.getElementById("speed").innerText=Math.floor(speed);
  document.getElementById("score").innerText=score;
 }

 renderer.render(scene,camera);
}

animate();

window.addEventListener("resize",()=>{
 camera.aspect=innerWidth/innerHeight;
 camera.updateProjectionMatrix();
 renderer.setSize(innerWidth,innerHeight);
});
